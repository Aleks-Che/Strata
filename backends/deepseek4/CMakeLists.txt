# Register the exact file identity and offset-zero mapping lifetime for native
# expert reads. Keep this out of upstream sources and the Qwen executable.
file(READ "${STRATA_LLAMA_DIR}/src/llama-mmap.cpp" mmap_source)
set(mmap_ctor "const ranges & lazy_ranges) : pimpl(std::make_unique<impl>(file, prefetch, numa, lazy_ranges)) {}")
set(mmap_dtor "llama_mmap::~llama_mmap() = default;")
foreach(anchor "${mmap_ctor}" "${mmap_dtor}")
  string(FIND "${mmap_source}" "${anchor}" found)
  if(found EQUAL -1)
    message(FATAL_ERROR "Pinned mmap lifecycle entry point not found")
  endif()
endforeach()
string(REPLACE "${mmap_ctor}" "const ranges & lazy_ranges) : pimpl(std::make_unique<impl>(file, prefetch, numa, lazy_ranges)) { strata_expert_file::add(addr(), size(), file->file_id()); }" mmap_source "${mmap_source}")
string(REPLACE "${mmap_dtor}" "llama_mmap::~llama_mmap() { strata_expert_file::remove(addr()); }" mmap_source "${mmap_source}")
string(PREPEND mmap_source "#include \"expert_file.hpp\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/strata-mmap.cpp" "${mmap_source}")

# Whole-context restore compacts rows starting at zero. Continue at its tail,
# as before saving, rather than immediately recycling old SWA cells at zero.
# Changing the physical order of keys changes DeepSeek's FP16 attention result.
file(READ "${STRATA_LLAMA_DIR}/src/llama-kv-cache.cpp" kv_source)
set(kv_anchor "        head = 0;\n    }\n\n    return true;\n}\n\nbool llama_kv_cache::state_read_data")
string(FIND "${kv_source}" "${kv_anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned whole-context KV restore tail not found")
endif()
string(REPLACE "${kv_anchor}" "        head = cell_count % cells.size();\n    }\n\n    return true;\n}\n\nbool llama_kv_cache::state_read_data" kv_source "${kv_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/strata-kv-cache.cpp" "${kv_source}")

# Enforce the GPU-only contract at the scheduler boundary, before executing any
# split. A missing CUDA implementation must fail instead of silently using CPU.
file(READ "${STRATA_LLAMA_DIR}/ggml/src/ggml-backend.cpp" scheduler)
set(audit_anchor "static enum ggml_status ggml_backend_sched_compute_splits(ggml_backend_sched_t sched) {")
string(FIND "${scheduler}" "${audit_anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned GGML scheduler entry point not found")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/gpu_only_audit.inc" audit)
file(READ "${CMAKE_CURRENT_LIST_DIR}/expert_pipeline_scope.inc" pipeline_scope)
string(REPLACE "${audit_anchor}" "${audit_anchor}\n${audit}\n${pipeline_scope}" scheduler "${scheduler}")
set(plan_anchor "                        prev_ids_tensor = ids_tensor;")
string(FIND "${scheduler}" "${plan_anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned GGML expert routing boundary not found")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/expert_pipeline_sched.inc" pipeline_plan)
string(REPLACE "${plan_anchor}" "${plan_anchor}\n${pipeline_plan}" scheduler "${scheduler}")
set(copy_anchor "auto copy_experts = [&](int32_t first_id, int32_t last_id) {")
string(FIND "${scheduler}" "${copy_anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned GGML selected-expert copy entry point not found")
endif()
set(copy_hook "\n auto reg = ggml_backend_dev_backend_reg(ggml_backend_get_device(split_backend));\n auto transfer = (StrataExpertCopy) ggml_backend_reg_get_proc_address(reg, \"strata_expert_copy\");\n if (transfer) { transfer(split_backend, input, input_cpy, first_id, last_id, int(ids_tensor->ne[1])); return; }\n")
set(prepare_hook "if(strata_direct_enabled) {\n auto reg=ggml_backend_dev_backend_reg(ggml_backend_get_device(split_backend));\n auto prepare=(StrataExpertPrepare)ggml_backend_reg_get_proc_address(reg, \"strata_expert_prepare\");\n if(prepare)prepare(split_backend,split->graph.nodes[0]);\n }\n ")
string(REPLACE "${copy_anchor}" "${prepare_hook}${copy_anchor}${copy_hook}" scheduler "${scheduler}")
set(copy_flush_anchor "                    copy_experts(first_id, last_id);\n                } else {")
string(FIND "${scheduler}" "${copy_flush_anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned GGML selected-expert copy tail not found")
endif()
string(REPLACE "${copy_flush_anchor}"
  "                    copy_experts(first_id, last_id);\n                    if(strata_copy_batch_enabled || strata_direct_enabled) {\n                        auto reg=ggml_backend_dev_backend_reg(ggml_backend_get_device(split_backend));\n                        auto flush=(StrataExpertFlush)ggml_backend_reg_get_proc_address(reg, \"strata_expert_flush\");\n                        if(flush)flush(split_backend);\n                    }\n                } else {"
  scheduler "${scheduler}")
string(PREPEND scheduler "#include \"expert_transfer.h\"\n")
string(PREPEND scheduler "#include \"shared_scratch.h\"\n")
string(APPEND scheduler "\n#include \"shared_scratch_sched.inc\"\n")
