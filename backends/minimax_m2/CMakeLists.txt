cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Configure MiniMax standalone in a separate build directory")
endif()
project(strata_minimax_m2 LANGUAGES C CXX)
option(STRATA_MM27_CUDA "Build isolated numerical CUDA checks" OFF)
option(STRATA_MM27_RUNTIME "Build experimental Windows selected-expert runtime" OFF)
option(STRATA_MM27_STRICT_F32 "Respect TF32 override in explicit CUDA MMF" ${STRATA_MM27_CUDA})
option(STRATA_MM27_ROUTED_STRIDES "Fix token-strided routed CUDA gather" ${STRATA_MM27_CUDA})
option(STRATA_MM27_QUANT_F32_CONTROL "Compile opt-in F32 activation reference for Q4_K/Q6_K" ${STRATA_MM27_CUDA})
if((STRATA_MM27_STRICT_F32 OR STRATA_MM27_ROUTED_STRIDES OR STRATA_MM27_QUANT_F32_CONTROL) AND NOT STRATA_MM27_CUDA)
  message(FATAL_ERROR "MiniMax CUDA patches require STRATA_MM27_CUDA=ON")
endif()
set(STRATA_MM27_ARCHIVE "" CACHE FILEPATH "Local archive of the audited candidate")
set(STRATA_MM27_SOURCE_SHA "86ebfef2c6a0f3359a2a07d2c215d61b0fa885c9")
set(STRATA_MM27_ARCHIVE_SHA256 "f8e524b635b726bae74fd8f84bb9249e5b09384c63207f6707c5a3f921acad99")
set(STRATA_MM27_LOADER_SHA256 "6574a8618655d8164588627c1a4d2fc4fa8953b65fd783b8c1849f1c5c93638e")
if(NOT EXISTS "${STRATA_MM27_ARCHIVE}" OR IS_DIRECTORY "${STRATA_MM27_ARCHIVE}")
  message(FATAL_ERROR "Set STRATA_MM27_ARCHIVE to the audited local archive")
endif()
file(SHA256 "${STRATA_MM27_ARCHIVE}" actual_hash)
if(NOT actual_hash STREQUAL STRATA_MM27_ARCHIVE_SHA256)
  message(FATAL_ERROR "MiniMax archive SHA-256 mismatch")
endif()
if(FETCHCONTENT_SOURCE_DIR_MM27_CANDIDATE)
  message(FATAL_ERROR "Use the verified archive, not an external source directory")
endif()
include(FetchContent)
set(FETCHCONTENT_BASE_DIR "${CMAKE_BINARY_DIR}/_deps" CACHE PATH "Private MiniMax dependency" FORCE)
FetchContent_Declare(mm27_candidate URL "${STRATA_MM27_ARCHIVE}"
  URL_HASH "SHA256=${STRATA_MM27_ARCHIVE_SHA256}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(mm27_candidate)
if(NOT mm27_candidate_POPULATED)
  FetchContent_Populate(mm27_candidate)
endif()
set(mm27_source "${mm27_candidate_SOURCE_DIR}")
file(SHA256 "${mm27_source}/src/models/minimax-m2.cpp" actual_loader_hash)
if(NOT actual_loader_hash STREQUAL STRATA_MM27_LOADER_SHA256)
  message(FATAL_ERROR "MiniMax loader differs from the audited candidate")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA ${STRATA_MM27_CUDA} CACHE BOOL "Isolated MiniMax CUDA backend" FORCE)
set(GGML_CUDA_FA_ALL_QUANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
if(STRATA_MM27_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_subdirectory("${mm27_source}" llama EXCLUDE_FROM_ALL)
set(mm27_patch_set "none")
set(mm27_cuda_json false)
if(STRATA_MM27_STRICT_F32)
  include(StrictF32.cmake)
endif()
if(STRATA_MM27_ROUTED_STRIDES)
  include(RoutedStrides.cmake)
endif()
if(STRATA_MM27_QUANT_F32_CONTROL)
  include(QuantF32.cmake)
endif()
if(STRATA_MM27_RUNTIME)
  if(NOT WIN32 OR NOT STRATA_MM27_CUDA OR NOT STRATA_MM27_STRICT_F32 OR
      NOT STRATA_MM27_ROUTED_STRIDES OR NOT STRATA_MM27_QUANT_F32_CONTROL)
    message(FATAL_ERROR "MiniMax runtime requires Windows CUDA and all numerical baseline patches")
  endif()
  include(RuntimePatches.cmake)
endif()

foreach(kind tokenizer loader)
  add_executable(strata-minimax-m2-${kind} ${kind}_oracle.cpp)
  target_include_directories(strata-minimax-m2-${kind} PRIVATE "${mm27_source}/vendor" "${mm27_source}/src")
  target_link_libraries(strata-minimax-m2-${kind} PRIVATE llama)
endforeach()
add_executable(strata-minimax-m2-template template_oracle.cpp "${mm27_source}/common/json.cpp"
  "${mm27_source}/common/unicode.cpp" "${mm27_source}/common/jinja/lexer.cpp"
  "${mm27_source}/common/jinja/parser.cpp" "${mm27_source}/common/jinja/runtime.cpp"
  "${mm27_source}/common/jinja/value.cpp" "${mm27_source}/common/jinja/string.cpp"
  "${mm27_source}/common/jinja/caps.cpp")
target_include_directories(strata-minimax-m2-template PRIVATE "${mm27_source}/common" "${mm27_source}/vendor")
target_link_libraries(strata-minimax-m2-template PRIVATE ggml-base)
enable_testing()
foreach(kind tokenizer loader template)
  set(target strata-minimax-m2-${kind})
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE STRATA_MM27_SOURCE_SHA="${STRATA_MM27_SOURCE_SHA}"
    STRATA_MM27_ARCHIVE_SHA256="${STRATA_MM27_ARCHIVE_SHA256}" STRATA_MM27_PATCH_SET="${mm27_patch_set}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /bigobj)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${kind}_oracle.cpp" mm27_${kind}_sha256)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${kind}_oracle.cpp")
  add_test(NAME minimax_m2_${kind}_version COMMAND ${target} --version)
  set_tests_properties(minimax_m2_${kind}_version PROPERTIES TIMEOUT 15)
endforeach()
if(STRATA_MM27_CUDA)
  set(mm27_cuda_json true)
  get_target_property(mm27_cuda_target_architectures ggml-cuda CUDA_ARCHITECTURES)
  foreach(kind kernels graph)
    set(target strata-minimax-m2-${kind}-check)
    add_executable(${target} check_${kind}.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${mm27_source}/vendor" "${mm27_source}/src")
    target_link_libraries(${target} PRIVATE llama)
    target_compile_definitions(${target} PRIVATE STRATA_MM27_SOURCE_SHA="${STRATA_MM27_SOURCE_SHA}"
      STRATA_MM27_ARCHIVE_SHA256="${STRATA_MM27_ARCHIVE_SHA256}" STRATA_MM27_PATCH_SET="${mm27_patch_set}")
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /bigobj)
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/check_${kind}.cpp" mm27_${kind}_sha256)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/check_${kind}.cpp")
  endforeach()
  set_property(SOURCE check_graph.cpp APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/synthetic_minimax_m2.hpp")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/synthetic_minimax_m2.hpp" mm27_fixture_sha256)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/synthetic_minimax_m2.hpp")
endif()
if(STRATA_MM27_RUNTIME)
  foreach(kind runtime-check lifecycle-check cache-check arena-check reload-check bench)
    if(kind STREQUAL "runtime-check")
      set(source check_runtime.cpp)
    elseif(kind STREQUAL "lifecycle-check")
      set(source check_lifecycle.cpp)
    elseif(kind STREQUAL "cache-check")
      set(source check_cache.cpp)
    elseif(kind STREQUAL "arena-check")
      set(source check_arena.cpp)
    elseif(kind STREQUAL "reload-check")
      set(source check_reload.cpp)
    else()
      set(source main.cpp)
    endif()
    set(target strata-minimax-m2-${kind})
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${mm27_source}/vendor" "${mm27_source}/src")
    target_link_libraries(${target} PRIVATE llama)
    if(kind STREQUAL "lifecycle-check" OR kind STREQUAL "arena-check" OR kind STREQUAL "reload-check")
      target_link_libraries(${target} PRIVATE CUDA::cudart_static)
    endif()
    target_compile_options(${target} PRIVATE /utf-8 /bigobj)
    target_compile_definitions(${target} PRIVATE STRATA_MM27_SOURCE_SHA="${STRATA_MM27_SOURCE_SHA}"
      STRATA_MM27_ARCHIVE_SHA256="${STRATA_MM27_ARCHIVE_SHA256}" STRATA_MM27_PATCH_SET="${mm27_patch_set}")
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set_property(SOURCE ${source} APPEND PROPERTY OBJECT_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/runtime.hpp;${CMAKE_CURRENT_SOURCE_DIR}/contract.hpp;${CMAKE_CURRENT_SOURCE_DIR}/sync_runtime.h;${CMAKE_CURRENT_SOURCE_DIR}/sync_test.h;${CMAKE_CURRENT_SOURCE_DIR}/synthetic_minimax_m2.hpp;${CMAKE_CURRENT_SOURCE_DIR}/../common/expert_file.hpp")
  endforeach()
endif()
configure_file(build_manifest.json.in "${CMAKE_BINARY_DIR}/minimax-m2-build-manifest.json" @ONLY)
