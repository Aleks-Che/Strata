cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Configure Hy3 as a standalone project in a separate build directory")
endif()
project(strata_hy3 LANGUAGES C CXX)
option(STRATA_HY3_MTP_LOAD_FLAGS "Skip all embedded MTP weights when load_mtp is false" ON)
option(STRATA_HY3_CUDA "Compile isolated CUDA candidate and numerical checks" OFF)
option(STRATA_HY3_RUNTIME "Compile native Hy3 pipe and synchronous selected-copy baseline" OFF)
option(STRATA_HY3_STRICT_F32 "Respect TF32 override in custom CUDA MMF dispatch" ${STRATA_HY3_CUDA})
option(STRATA_HY3_ROUTED_STRIDES "Fix physical row indexing in routed CUDA gather" ${STRATA_HY3_CUDA})
if((STRATA_HY3_STRICT_F32 OR STRATA_HY3_ROUTED_STRIDES) AND NOT STRATA_HY3_CUDA)
  message(FATAL_ERROR "Hy3 CUDA patches require STRATA_HY3_CUDA=ON")
endif()
if(STRATA_HY3_RUNTIME AND (NOT WIN32 OR NOT STRATA_HY3_CUDA OR NOT STRATA_HY3_STRICT_F32 OR NOT STRATA_HY3_ROUTED_STRIDES OR NOT STRATA_HY3_MTP_LOAD_FLAGS))
  message(FATAL_ERROR "Hy3 runtime requires Windows, CUDA and all three correctness patches")
endif()
set(STRATA_HY3_ARCHIVE "" CACHE FILEPATH "Local archive of the audited Hy3 candidate")
set(STRATA_HY3_SOURCE_SHA "86ebfef2c6a0f3359a2a07d2c215d61b0fa885c9")
set(STRATA_HY3_ARCHIVE_SHA256 "f8e524b635b726bae74fd8f84bb9249e5b09384c63207f6707c5a3f921acad99")
set(STRATA_HY3_LOADER_SHA256 "224921b8ce6f9be02dc1252ef6847a93f81386021b24d7d08eb25444e787d62e")
if(NOT EXISTS "${STRATA_HY3_ARCHIVE}" OR IS_DIRECTORY "${STRATA_HY3_ARCHIVE}")
  message(FATAL_ERROR "Set STRATA_HY3_ARCHIVE to the local ${STRATA_HY3_SOURCE_SHA}.tar.gz archive")
endif()
file(SHA256 "${STRATA_HY3_ARCHIVE}" actual_hash)
if(NOT actual_hash STREQUAL STRATA_HY3_ARCHIVE_SHA256)
  message(FATAL_ERROR "Hy3 archive SHA-256 mismatch; audit a new candidate separately")
endif()
if(FETCHCONTENT_SOURCE_DIR_HY3_CANDIDATE)
  message(FATAL_ERROR "Use the verified archive, not FETCHCONTENT_SOURCE_DIR_HY3_CANDIDATE")
endif()
include(FetchContent)
set(FETCHCONTENT_BASE_DIR "${CMAKE_BINARY_DIR}/_deps" CACHE PATH "Private Hy3 dependency" FORCE)
FetchContent_Declare(hy3_candidate URL "${STRATA_HY3_ARCHIVE}"
  URL_HASH "SHA256=${STRATA_HY3_ARCHIVE_SHA256}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(hy3_candidate)
if(NOT hy3_candidate_POPULATED)
  FetchContent_Populate(hy3_candidate)
endif()
set(hy3_source "${hy3_candidate_SOURCE_DIR}")
file(SHA256 "${hy3_source}/src/models/hy-v3.cpp" actual_loader_hash)
if(NOT actual_loader_hash STREQUAL STRATA_HY3_LOADER_SHA256)
  message(FATAL_ERROR "Hy3 loader differs from the audited candidate")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA ${STRATA_HY3_CUDA} CACHE BOOL "Isolated Hy3 CUDA" FORCE)
set(GGML_CUDA_FA_ALL_QUANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
if(STRATA_HY3_CUDA)
  enable_language(CUDA)
endif()
add_subdirectory("${hy3_source}" llama EXCLUDE_FROM_ALL)
set(hy3_patch_set "none")
set(hy3_generated_sha256 "${STRATA_HY3_LOADER_SHA256}")
if(STRATA_HY3_MTP_LOAD_FLAGS)
  include(MtpLoadFlags.cmake)
endif()
if(STRATA_HY3_STRICT_F32)
  include(StrictF32.cmake)
endif()
if(STRATA_HY3_ROUTED_STRIDES)
  include(RoutedStrides.cmake)
endif()
if(STRATA_HY3_RUNTIME)
  find_package(CUDAToolkit REQUIRED)
  include(RuntimePatches.cmake)
endif()
add_executable(strata-hy3-tokenizer tokenizer_oracle.cpp)
target_include_directories(strata-hy3-tokenizer PRIVATE "${hy3_source}/vendor")
target_link_libraries(strata-hy3-tokenizer PRIVATE llama)
add_executable(strata-hy3-loader loader_oracle.cpp)
target_include_directories(strata-hy3-loader PRIVATE "${hy3_source}/vendor" "${hy3_source}/src")
target_link_libraries(strata-hy3-loader PRIVATE llama)
add_executable(strata-hy3-template template_oracle.cpp "${hy3_source}/common/json.cpp"
  "${hy3_source}/common/unicode.cpp"
  "${hy3_source}/common/jinja/lexer.cpp" "${hy3_source}/common/jinja/parser.cpp"
  "${hy3_source}/common/jinja/runtime.cpp" "${hy3_source}/common/jinja/value.cpp"
  "${hy3_source}/common/jinja/string.cpp" "${hy3_source}/common/jinja/caps.cpp")
target_include_directories(strata-hy3-template PRIVATE "${hy3_source}/common" "${hy3_source}/vendor")
target_link_libraries(strata-hy3-template PRIVATE ggml-base)
foreach(target strata-hy3-tokenizer strata-hy3-template strata-hy3-loader)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE STRATA_HY3_SOURCE_SHA="${STRATA_HY3_SOURCE_SHA}"
    STRATA_HY3_ARCHIVE_SHA256="${STRATA_HY3_ARCHIVE_SHA256}" STRATA_HY3_PATCH_SET="${hy3_patch_set}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /bigobj)
  endif()
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
set(hy3_cuda_json false)
if(STRATA_HY3_CUDA)
  set(hy3_cuda_json true)
  add_executable(strata-hy3-kernels-check check_kernels.cpp)
  add_executable(strata-hy3-graph-check check_graph.cpp)
  set(hy3_cuda_targets strata-hy3-kernels-check strata-hy3-graph-check)
  if(STRATA_HY3_RUNTIME)
    add_executable(strata-hy3 main.cpp)
    add_executable(strata-hy3-runtime-check check_runtime.cpp)
    # Explicit header dependencies also cover localized MSVC /showIncludes output.
    set_property(SOURCE main.cpp check_runtime.cpp APPEND PROPERTY OBJECT_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/runtime.hpp;${CMAKE_CURRENT_SOURCE_DIR}/contract.hpp;${CMAKE_CURRENT_SOURCE_DIR}/sync_runtime.h;${CMAKE_CURRENT_SOURCE_DIR}/sync_test.h;${CMAKE_CURRENT_SOURCE_DIR}/protocol.hpp;${CMAKE_CURRENT_SOURCE_DIR}/tokenizer_protocol.hpp;${CMAKE_CURRENT_SOURCE_DIR}/synthetic_hy3.hpp;${CMAKE_CURRENT_SOURCE_DIR}/../common/expert_file.hpp")
    list(APPEND hy3_cuda_targets strata-hy3 strata-hy3-runtime-check)
  endif()
  foreach(target ${hy3_cuda_targets})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${hy3_source}/vendor" "${hy3_source}/src")
    target_link_libraries(${target} PRIVATE llama)
    target_compile_definitions(${target} PRIVATE STRATA_HY3_SOURCE_SHA="${STRATA_HY3_SOURCE_SHA}"
      STRATA_HY3_ARCHIVE_SHA256="${STRATA_HY3_ARCHIVE_SHA256}" STRATA_HY3_PATCH_SET="${hy3_patch_set}")
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /bigobj)
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  set_property(SOURCE check_graph.cpp APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/synthetic_hy3.hpp")
endif()
configure_file(build_manifest.json.in "${CMAKE_BINARY_DIR}/hy3-build-manifest.json" @ONLY)
enable_testing()
add_test(NAME hy3_tokenizer_version COMMAND strata-hy3-tokenizer --version)
add_test(NAME hy3_template_version COMMAND strata-hy3-template --version)
set_tests_properties(hy3_tokenizer_version hy3_template_version PROPERTIES TIMEOUT 15)
